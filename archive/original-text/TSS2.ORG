%ROUTINE TO INITIALIZE THE STACK

       SAA 0; STA -1,X; SAA -1; STA 0,X
       LDA (STK0+STLEN-12; STA I 9ESTK
       EXIT

%ROUTINE TO STACK THE REGISTERS AND ALLOCATE SPACE
%FOR LOCAL VARIABLES
%SAVA = SAVED A-REGISTER
%RETURN (X-REGISTER CONTAINS POINTER TO ENTRY)

STK,   SWAP DA SL; STA I 9SAVL; STX I 9SAVX; STT I 9SAVT
       SAA 2; RADD DL SA
       LDX I 9SPTR; LDA -1,X; RADD DA SX AD1; AAA 12
       SWAP DX SA; STA 0,X; STX I 9SPTR
       COPY DX SL; LDA -1,X; LDX I 9SPTR
       STA -1,X; RADD DA SX; AAA 12
       LDT I 9ESTK; SKP DT LST SA; JMP *+2; JMP STKER
       LDA I (SAVA; LDT I 9SAVT; STF TREG,X
       LDA I 9SAVX; COPY DD SA; LDA I 9SAVL
       COPY DT SB; STT XBREG,X
       LDT (RBLOK; COPY DB ST; STF BREG,X
       LDA I 9OVLX; STA XOVL,X; LDA I 9OVL; STA I 9OVLX
       LDF TREG,X
       EXIT
STKER, JMP I (STKOV
       )FILL
       )KILL STKER
